function isMarket(
  value: string | null,
): value is Market {
  return value === "US" || value === "KR";
}

function isRequestedGender(
  value: string | null,
): value is Exclude<Gender, "UNISEX"> {
  return value === "MEN" || value === "WOMEN";
}

function parsePrice(
  value?: string,
): number | null {
  if (!value) {
    return null;
  }

  const normalized = value
    .replace(/,/g, "")
    .replace(/[^\d.]/g, "");

  const result = Number.parseFloat(normalized);

  return Number.isFinite(result)
    ? result
    : null;
}

function getDiscountFromText(
  result: ShoppingResult,
): number | null {
  const values = [
    result.tag,
    result.badge,
    ...(result.extensions ?? []),
  ].filter(
    (value): value is string =>
      Boolean(value),
  );

  for (const value of values) {
    const match =
      value.match(
        /(\d{1,2})\s*%\s*(?:off|할인)?/i,
      ) ??
      value.match(
        /(?:할인율\s*)?(\d{1,2})\s*%/i,
      );

    if (!match) {
      continue;
    }

    const percentage =
      Number.parseInt(match[1], 10);

    if (
      percentage > 0 &&
      percentage < 100
    ) {
      return percentage;
    }
  }

  return null;
}

function resolveSalePrices(
  result: ShoppingResult,
): {
  salePrice: number;
  originalPrice: number;
  discountPercentage: number;
} | null {
  const salePrice =
    result.extracted_price ??
    parsePrice(result.price);

  if (
    salePrice === null ||
    !Number.isFinite(salePrice) ||
    salePrice <= 0
  ) {
    return null;
  }

  const oldPrice =
    result.extracted_old_price ??
    parsePrice(result.old_price);

  if (
    oldPrice !== null &&
    Number.isFinite(oldPrice) &&
    oldPrice > salePrice
  ) {
    return {
      salePrice,
      originalPrice: oldPrice,
      discountPercentage:
        calculateDiscountPercentage(
          oldPrice,
          salePrice,
        ),
    };
  }

  const explicitDiscount =
    getDiscountFromText(result);

  if (!explicitDiscount) {
    return null;
  }

  const calculatedOriginalPrice =
    salePrice /
    (1 - explicitDiscount / 100);

  if (
    !Number.isFinite(
      calculatedOriginalPrice,
    ) ||
    calculatedOriginalPrice <= salePrice
  ) {
    return null;
  }

  return {
    salePrice,
    originalPrice:
      Math.round(
        calculatedOriginalPrice / 10,
      ) * 10,
    discountPercentage:
      explicitDiscount,
  };
}

function createProductId(
  result: ShoppingResult,
): string {
  if (result.product_id) {
    return result.product_id;
  }

  return createHash("sha256")
    .update(
      [
        result.title ?? "",
        result.source ?? "",
        result.link ??
          result.product_link ??
          "",
      ].join("|"),
    )
    .digest("hex")
    .slice(0, 24);
}

function collectResultGroups(
  response: SerpApiResponse,
): ShoppingResult[] {
  const categorized =
    response
      .categorized_shopping_results
      ?.flatMap(
        (group) =>
          group.shopping_results ?? [],
      ) ?? [];

  return [
    ...(response.shopping_results ?? []),
    ...(response
      .inline_shopping_results ?? []),
    ...categorized,
  ];
}

function createQuery(
  brand: BrandSearch,
  market: Market,
  gender: Exclude<
    Gender,
    "UNISEX"
  >,
): string {
  if (market === "KR") {
    const genderTerm =
      gender === "MEN"
        ? "남성"
        : "여성";

    /*
     * Do not require both English and Korean
     * brand names. OR increases recall.
     */
    return `("${brand.englishName}" OR "${brand.koreanName}") ${genderTerm}`;
  }

  const genderTerm =
    gender === "MEN"
      ? "men"
      : "women";

  return `"${brand.englishName}" ${genderTerm}`;
}

function normalizeProduct(
  result: ShoppingResult,
  brand: BrandSearch,
  market: Market,
  requestedGender: Exclude<
    Gender,
    "UNISEX"
  >,
): Product | null {
  const title =
    result.title?.trim();

  const sourceName =
    result.source?.trim() ||
    "Online retailer";

  const sourceUrl =
    result.link?.trim() ??
    result.product_link?.trim();

  const imageUrl =
    result.serpapi_thumbnail?.trim() ??
    result.thumbnail?.trim();

  if (
    !title ||
    !sourceUrl ||
    !imageUrl
  ) {
    return null;
  }

  if (result.second_hand_condition) {
    return null;
  }

  const prices =
    resolveSalePrices(result);

  if (!prices) {
    return null;
  }

  /*
   * Search results may include sponsored or loosely related products.
   * Confirm the brand from the actual listing before returning it.
   */
  const detectedBrand = inferBrand(
    [
      title,
      result.source,
      result.tag,
      result.badge,
      ...(result.extensions ?? []),
    ]
      .filter(
        (value): value is string =>
          Boolean(value),
      )
      .join(" "),
  );

  if (
    !detectedBrand ||
    detectedBrand.slug !== brand.slug
  ) {
    return null;
  }

  return {
    id: createProductId(result),

    sourceName,
    sourceUrl,

    market,
    currency:
      market === "KR"
        ? "KRW"
        : "USD",

    gender: inferGender(
      title,
      requestedGender,
    ),

    /*
     * The search itself is brand-specific,
     * so do not require the title to contain
     * the brand again.
     */
    /*
     * Use the brand detected from the listing itself.
     */
    brandSlug: detectedBrand.slug,
    brandName: detectedBrand.name,

    productName: title,
    categorySlug:
      inferCategory(title),

    imageUrl,

    originalPrice:
      prices.originalPrice,
    salePrice:
      prices.salePrice,
    discountPercentage:
      prices.discountPercentage,
    saleVerification:
      prices.originalPrice !== null &&
      prices.discountPercentage !== null
        ? "VERIFIED"
        : "UNVERIFIED",

    isAvailable: true,
    collectedAt:
      new Date().toISOString(),
  };
}

async function fetchSearchPage(
  url: string,
): Promise<SearchPageResult> {
  try {
    const response = await fetch(url, {
      headers: {
        Accept: "application/json",
      },
      next: {
        revalidate: 3600,
      },
    });

    const body =
      (await response.json()) as SerpApiResponse;

    if (!response.ok || body.error) {
      return {
        results: [],
        error:
          body.error ??
          `HTTP ${response.status}`,
      };
    }

    return {
      results:
        collectResultGroups(body),
      nextUrl:
        body.serpapi_pagination?.next,
    };
  } catch (error) {
    return {
      results: [],
      error:
        error instanceof Error
          ? error.message
          : "Unknown request error",
    };
  }
}

"use client";

import useSWR from "swr";
import type {
  Gender,
  Market,
  Product,
} from "@/types/product";

interface ProductResponse {
  products: Product[];
  count: number;
  market: Market;
  gender: Gender;
  updatedAt: string;
}

interface ErrorResponse {
  error?: string;
}

async function fetcher(
  url: string,
): Promise<ProductResponse> {
  const response = await fetch(url);

  const body = (await response.json()) as
    | ProductResponse
    | ErrorResponse;

  if (!response.ok) {
    throw new Error(
      "error" in body && body.error
        ? body.error
        : "Unable to load product listings.",
    );
  }

  return body as ProductResponse;
}

export function useProducts(
  market: Market,
  gender: Exclude<Gender, "UNISEX">,
) {
  const query = new URLSearchParams({
    market,
    gender,
  });

  return useSWR<ProductResponse>(
    `/api/products?${query.toString()}`,
    fetcher,
    {
      revalidateOnFocus: false,
      dedupingInterval: 60 * 60 * 1000,
      keepPreviousData: true,
    },
  );
}
