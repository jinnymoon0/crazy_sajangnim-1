"use client";

import useSWR from "swr";

import type { Gender, Market, Product } from "@/types/product";

interface ProductResponse {
  products: Product[];

  page: number;
  size: number;
  total: number;
  hasMore: boolean;

  market: Market;
  gender: Exclude<Gender, "UNISEX">;

  updatedAt: string;
}

interface ErrorResponse {
  error?: string;
}

async function fetcher(url: string): Promise<ProductResponse> {
  const response = await fetch(url);

  const body = (await response.json()) as ProductResponse | ErrorResponse;

  if (!response.ok) {
    throw new Error(
      "error" in body && body.error
        ? body.error
        : "Unable to load product listings.",
    );
  }

  return body as ProductResponse;
}

export function useProducts(market: Market, gender: Exclude<Gender, "UNISEX">) {
  const query = new URLSearchParams({
    market,
    gender,
    page: "1",
    size: "300",
    sort: "discount-desc",
  });

  return useSWR<ProductResponse>(
    `/api/catalog/products?${query.toString()}`,
    fetcher,
    {
      revalidateOnFocus: false,
      revalidateOnReconnect: false,
      dedupingInterval: 5 * 60 * 1000,
      keepPreviousData: true,
    },
  );
}

import { NextRequest, NextResponse } from "next/server";

import { getProductsPage } from "@/lib/products/product-repository";

import type { Gender, Market, ProductSort } from "@/types/product";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

function isMarket(value: string | null): value is Market {
  return value === "US" || value === "KR";
}

function isGender(value: string | null): value is Exclude<Gender, "UNISEX"> {
  return value === "MEN" || value === "WOMEN";
}

function isSort(value: string | null): value is ProductSort {
  return (
    value === "discount-desc" ||
    value === "price-asc" ||
    value === "price-desc" ||
    value === "newest"
  );
}

function parsePositiveInteger(value: string | null, fallback: number): number {
  const parsed = Number.parseInt(value ?? "", 10);

  return Number.isFinite(parsed) && parsed > 0 ? parsed : fallback;
}

export async function GET(request: NextRequest) {
  try {
    const parameters = request.nextUrl.searchParams;

    const marketValue = parameters.get("market");

    const genderValue = parameters.get("gender");

    const sortValue = parameters.get("sort");

    const market: Market = isMarket(marketValue) ? marketValue : "US";

    const gender: Exclude<Gender, "UNISEX"> = isGender(genderValue)
      ? genderValue
      : "MEN";

    const sort: ProductSort = isSort(sortValue) ? sortValue : "discount-desc";

    const page = parsePositiveInteger(parameters.get("page"), 1);

    const size = Math.min(
      300,
      parsePositiveInteger(parameters.get("size"), 60),
    );

    const brandSlugs = parameters.getAll("brand").filter(Boolean);

    const categorySlug = parameters.get("category");

    const result = await getProductsPage({
      market,
      gender,
      page,
      size,
      sort,
      brandSlugs: brandSlugs as Parameters<
        typeof getProductsPage
      >[0]["brandSlugs"],
      categorySlug:
        categorySlug && categorySlug !== "all"
          ? (categorySlug as Parameters<
              typeof getProductsPage
            >[0]["categorySlug"])
          : null,
    });

    return NextResponse.json(result, {
      headers: {
        "Cache-Control": "public, s-maxage=300, stale-while-revalidate=900",
      },
    });
  } catch (error) {
    console.error("Product catalog read failed:", error);

    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "Unable to load product catalog.",
      },
      {
        status: 500,
      },
    );
  }
}

export interface ProductCatalogPage {
  products: Product[];

  page: number;
  size: number;
  total: number;
  hasMore: boolean;

  market: Market;
  gender: Exclude<Gender, "UNISEX">;

function createProductKey(product: Product): string {
  return createHash("sha256")
    .update(
      [
        product.market,
        product.gender,
        product.brandSlug,
        product.sourceUrl.trim(),
        product.productName.trim(),
      ].join("|"),
    )
    .digest("hex");
}

function productToRow(product: Product): ProductCatalogRow {
  const now = new Date().toISOString();

  return {
    product_key: createProductKey(product),

    market: product.market,
    currency: product.currency,
    gender: product.gender,

    brand_slug: product.brandSlug,
    brand_name: product.brandName,

    product_name: product.productName,
    category_slug: product.categorySlug,

    source_name: product.sourceName,
    source_url: product.sourceUrl,
    image_url: product.imageUrl,

    original_price: product.originalPrice,
    sale_price: product.salePrice,
    discount_percentage: product.discountPercentage,

    sale_verification: product.saleVerification,

    is_available: product.isAvailable,

    collected_at: product.collectedAt,
    last_seen_at: now,
  };
}

function rowToProduct(row: ProductCatalogRow): Product {
  return {
    id: row.product_key.slice(0, 24),

    sourceName: row.source_name,
    sourceUrl: row.source_url,

    market: row.market,
    currency: row.currency,
    gender: row.gender,

    brandSlug: row.brand_slug,
    brandName: row.brand_name,

    productName: row.product_name,
    categorySlug: row.category_slug,

    imageUrl: row.image_url,

    originalPrice:
      row.original_price === null ? null : Number(row.original_price),

    salePrice: Number(row.sale_price),

    discountPercentage:
      row.discount_percentage === null ? null : Number(row.discount_percentage),

    saleVerification: row.sale_verification,

    isAvailable: row.is_available,
    collectedAt: row.collected_at,
  };
}

export async function upsertProducts(products: Product[]): Promise<number> {
  if (products.length === 0) {
    return 0;
  }

  const supabase = getSupabaseAdmin();

  const uniqueRows = Array.from(
    new Map(
      products.map((product) => {
        const row = productToRow(product);

        return [row.product_key, row];
      }),
    ).values(),
  );

  const batchSize = 500;
  let savedCount = 0;

  for (let index = 0; index < uniqueRows.length; index += batchSize) {
    const batch = uniqueRows.slice(index, index + batchSize);

    const { error } = await supabase.from(PRODUCT_TABLE).upsert(batch, {
      onConflict: "product_key",
      ignoreDuplicates: false,
    });

    if (error) {
      throw new Error(`Unable to save product catalog: ${error.message}`);
    }

    savedCount += batch.length;
  }

  return savedCount;
}

export async function getProductsPage({
  market,
  gender,
  page,
  size,
  sort = "discount-desc",
  brandSlugs = [],
  categorySlug = null,
}: ProductCatalogQuery): Promise<ProductCatalogPage> {
  const supabase = getSupabaseAdmin();

  const safePage = Math.max(1, page);
  const safeSize = Math.min(300, Math.max(1, size));

  const start = (safePage - 1) * safeSize;

  const end = start + safeSize - 1;

  let query = supabase
    .from(PRODUCT_TABLE)
    .select(
      [
        "product_key",
        "market",
        "currency",
        "gender",
        "brand_slug",
        "brand_name",
        "product_name",
        "category_slug",
        "source_name",
        "source_url",
        "image_url",
        "original_price",
        "sale_price",
        "discount_percentage",
        "sale_verification",
        "is_available",
        "collected_at",
        "first_seen_at",
        "last_seen_at",
      ].join(","),
      {
        count: "exact",
      },
    )
    .eq("market", market)
    .in("gender", [gender, "UNISEX"])
    .eq("is_available", true)
    .eq("sale_verification", "VERIFIED");

  if (brandSlugs.length > 0) {
    query = query.in("brand_slug", brandSlugs);
  }

  if (categorySlug) {
    query = query.eq("category_slug", categorySlug);
  }

  switch (sort) {
    case "price-asc":
      query = query.order("sale_price", {
        ascending: true,
      });
      break;

    case "price-desc":
      query = query.order("sale_price", {
        ascending: false,
      });
      break;

    case "newest":
      query = query.order("last_seen_at", {
        ascending: false,
      });
      break;

    case "discount-desc":
    default:
      query = query
        .order("discount_percentage", {
          ascending: false,
          nullsFirst: false,
        })
        .order("last_seen_at", {
          ascending: false,
        });
      break;
  }

  const { data, error, count } = await query.range(start, end);

  if (error) {
    throw new Error(`Unable to read product catalog: ${error.message}`);
  }

  const rows = (data ?? []) as unknown as ProductCatalogRow[];

  const total = count ?? 0;

  return {
    products: rows.map(rowToProduct),

    page: safePage,
    size: safeSize,
    total,
    hasMore: end + 1 < total,

    market,
    gender,

    updatedAt: new Date().toISOString(),
  };
}

export async function getCatalogCount(
  market: Market,
  gender: Exclude<Gender, "UNISEX">,
): Promise<number> {
  const supabase = getSupabaseAdmin();

  const { count, error } = await supabase
    .from(PRODUCT_TABLE)
    .select("product_key", {
      count: "exact",
      head: true,
    })
    .eq("market", market)
    .in("gender", [gender, "UNISEX"])
    .eq("is_available", true)
    .eq("sale_verification", "VERIFIED");

  if (error) {
    throw new Error(`Unable to count products: ${error.message}`);
  }

  return count ?? 0;
}
